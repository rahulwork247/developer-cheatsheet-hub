Python 3.14 Async Pattern

import asyncio

async def worker(id):
    print(f"Worker {id} starting")
    await asyncio.sleep(1)
    print(f"Worker {id} done")

async def main():
    await asyncio.gather(*(worker(i) for i in range(5)))

if __name__ == "__main__":
    asyncio.run(main())
