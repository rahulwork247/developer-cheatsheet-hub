Docker Commands

# List containers
docker ps -a

Python Async Patterns (3.11+)

import asyncio

async def worker(id):
    await asyncio.sleep(1)
    print(f"worker {id} done")

async def main():
    await asyncio.gather(*(worker(i) for i in range(5)))

asyncio.run(main())
